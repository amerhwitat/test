cmake_minimum_required(VERSION 3.20)
project(aurora_compositor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Quick WaylandCompositor)
qt_standard_project_setup()
qt_add_executable(aurora-compositor main.cpp)
qt_add_qml_module(aurora-compositor URI Aurora VERSION 1.0 QML_FILES shell.qml)
target_link_libraries(aurora-compositor PRIVATE Qt6::Gui Qt6::Quick Qt6::WaylandCompositor)
